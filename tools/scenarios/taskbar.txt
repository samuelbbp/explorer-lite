# The taskbar's settings. The resident's memory alone, with the settings window open and after closing it (Direct2D
# goes with the window). Then the bar's own menu (a right click on its empty space) in a picture, and its rows: the
# settings window and the task manager open from it. Each layout applied as a click in the window applies it (the
# resident saves, the shell reads again), checked by the bar itself (barlayout) and in a picture: left, centered,
# Start kept at the far left, a custom Start icon (.ico), small with seconds and icons only, the main display's bar only.
conf language=en
conf pinned=C:\Windows\System32\notepad.exe|C:\Windows\regedit.exe
start --tray
sleep 2000
shell
sleep 3000
shellmem shell-alone
# the first update check (30 s after start) loads winhttp for good: done before, so the numbers are the window's
sleep 30000
mem resident-alone
send --taskbar-settings
sleep 2000
mem resident-settings-open
png settingsshot settings-dark
cmd settingsclose
sleep 3000
mem resident-after-settings
send C:\Windows
sleep 1500
send --tasks
sleep 3000
shellcmd barmenu
sleep 500
shellpng barmenushot menu-dark
shellcmd barmenupick 1
sleep 2000
logcheck taskbar settings: open
shellcmd barmenu
sleep 300
shellcmd barmenupick 0
sleep 1500
shellcmd barlayout
logcheck barlayout: not centered, start at the left end, height 40, start icon logo
shellshot bar-left-dark
cmd settingsset align 1
sleep 1000
logcheck shell: taskbar settings applied
shellcmd barlayout
logcheck barlayout: centered, start with the apps
shellshot bar-centered-dark
cmd settingsset start 1
sleep 1000
shellcmd barlayout
logcheck barlayout: centered, start at the left end
shellshot bar-centered-start-left-dark
cmd settingsset start 0
cmd settingsset icon {dir}\explorer-lite.ico
sleep 1000
shellcmd barlayout
logcheck start with the apps, height 40, start icon custom
shellshot bar-custom-icon-dark
png settingsshot settings-custom-dark
cmd settingsset size 1
cmd settingsset seconds 1
cmd settingsset labels 0
sleep 1000
shellcmd barlayout
logcheck height 32, start icon custom
shellshot bar-small-dark
cmd settingsset displays 0
sleep 1500
shellcmd barlayout
logcheck 1 bars
shellshot bar-one-display-dark
png settingsshot settings-changed-dark
shellmem shell-after-settings
cmd settingsclose
stop
