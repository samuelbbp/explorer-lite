# The shell: a taskbar on each monitor with the open windows (explorer-lite's own, the only ones on the test desktop),
# two pinned apps and the clock; the window filter checked on real windows of each kind; Start with all apps, then a
# search typed. The shell's memory alone, with windows, with Start open and after closing it. Then a crash (it starts
# again by itself) and the installer's way: --quit closes the shell too, --updated brings it back. Nothing is launched.
conf language=en
conf pinned=C:\Windows\System32\notepad.exe|C:\Windows\regedit.exe
start --tray
sleep 2000
shell
sleep 3000
shellmem shell-alone
send C:\Windows
sleep 1500
send --tasks
sleep 3000
shellcmd filtertest
sleep 1500
logcheck filtertest: ok
shellcmd tasks
sleep 500
shellmem shell-with-windows
shellshot bar-dark
shellcmd start
sleep 3000
shellmem shell-start-open
startshot start-dark
shellcmd search no
sleep 1500
startshot start-search-dark
shellcmd startclose
sleep 3000
shellmem shell-after-start
shellcmd crash
sleep 7000
logcheck shell: crashed, starting again
logcheck after a crash
send --quit
sleep 2000
start --updated --tray
sleep 4000
logcheck shell: started again after the update
stop
